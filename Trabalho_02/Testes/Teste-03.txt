#include <iostream>
#include <string>
#include <iomanip>
#include <algorithm>
#include "TriangleMesh.h"
#include "Mesh.h"
#include "HEMeshDecoration.h"
#include <random>

namespace tcii::cg {
    ObjectPtr<TriangleMesh> readOBJ(const char* filename);
}

using namespace tcii::cg;

// Definição dos tipos de decoração
// 0: Vértices (Cor - Vec3f), 1: Arestas (Peso - float), 2: Faces (Flags - int), 3: Bordas
using VDeco = Vec3f; 
using EDeco = float;
using FDeco = int;
using BDeco = int;

int main(int argc, char** argv) {
    const char* filename = "f-16.obj";

    auto triangleMeshPtr = readOBJ(filename);
    if (!triangleMeshPtr) return -1;

    Mesh myMesh(*triangleMeshPtr);

    // Inicialização da Decoração
    auto decoration = HEMeshDecoration<VDeco, EDeco, FDeco, BDeco>::New(
        myMesh.vertexCount(), myMesh.edgeCount(), myMesh.faceCount(), myMesh.boundaryCount()
    );

    // Preenchimento de exemplo
    std::random_device aleatorio; 
    std::mt19937 floats_aleatorios(aleatorio()); 
    std::uniform_real_distribution<float> intervalo(0.0f, 1.0f);

    for (index_t i = 0; i < myMesh.vertexCount(); ++i)
    { 
        float r = intervalo(floats_aleatorios);
        float g = intervalo(floats_aleatorios);
        float b = intervalo(floats_aleatorios);
        decoration->setAttr<0>(i, Vec3f(r,g,b));
    }
    for (index_t i = 0; i < myMesh.faceCount(); ++i)   decoration->setAttr<2>(i, (i % 2));

    myMesh.setDecoration(ObjectPtr<SharedObject>(decoration.get()));

    // Exibição dos Dados (Estilo teste_01.txt)
    auto deco = myMesh.getDecoration<HEMeshDecoration<VDeco, EDeco, FDeco, BDeco>>();
    
    std::cout << "---DECORAÇÃO (A3) ---" << std::endl;
    std::cout << "Vértices: " << myMesh.vertexCount() << " | Faces: " << myMesh.faceCount() << " | Semi-Arestas:  " << (myMesh.edgeCount() * 2) << " | Bordas: " << myMesh.boundaryCount() << std::endl;

    std::cout << "\n[VÉRTICES - Cor]" << std::endl;
    for (index_t i = 0; i < std::min(myMesh.vertexCount(), 10u); ++i) {
        Vec3f cor = deco->getAttr<0>(i);
        std::cout << "V" << i << " | Cor: " << cor.x << "," << cor.y << "," << cor.z << std::endl;
    }
    std::cout << "E mais 2334 vértices..." << std::endl;

    std::cout << "\n[FACES - Flag (Par ou Ímpar)]" << std::endl;
    for (index_t i = 0; i < std::min(myMesh.faceCount(), 10u); ++i) {
        std::cout << "F" << i << " | Flag: " << deco->getAttr<2>(i) << std::endl;
    }
    std::cout << "E mais 4582 faces..." << std::endl;

    return 0;
 }
